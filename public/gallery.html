async function fetchAndPlayMotion() { if (!currentSelectedSnap) return; // 1. Strip image extension (.jpg, .png, etc.) const baseName = currentSelectedSnap.filename.replace(/\.(jpe?g|png|webp)$/i, ''); // 2. Use the exact base name with .mp4 extension (No extra '_motion' suffix) const mp4File = `${baseName}.mp4`; const playBtn = document.getElementById('modal-play-action'); playBtn.disabled = true; playBtn.textContent = 'Checking cloud cache…'; try { const checkRes = await fetch(`/api/motion/check?location=${LOCATION_CODE}&cam=${encodeURIComponent(CAMERA)}&file=${encodeURIComponent(mp4File)}`); const statusData = await checkRes.json(); if (statusData.status === 'ready') { startPlayback(mp4File); return; } playBtn.textContent = 'Fetching from camera disk…'; let attempts = 0; const poll = setInterval(async () => { attempts++; const retryRes = await fetch(`/api/motion/check?location=${LOCATION_CODE}&cam=${encodeURIComponent(CAMERA)}&file=${encodeURIComponent(mp4File)}`); const checkAgain = await retryRes.json(); if (checkAgain.status === 'ready') { clearInterval(poll); startPlayback(mp4File); } else if (attempts > 15) { clearInterval(poll); playBtn.textContent = 'File not found on local disk'; } }, 1500); } catch (err) { playBtn.textContent = 'Connection error'; playBtn.disabled = false; } }
