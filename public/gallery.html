def process_on_demand_requests(): request_prefix = f"{LOCATION_ID}/requests/" logger.info(f"On-demand motion request listener active on prefix: {request_prefix}") while True: try: res = s3_client.list_objects_v2( Bucket=R2_BUCKET_NAME, Prefix=request_prefix, MaxKeys=10 ) for item in res.get("Contents", []): marker_key = item["Key"] if not marker_key.endswith(".req"): continue filename_part = marker_key[len(request_prefix):][:-4] # Remove prefix & '.req' if "__" in filename_part: cam_name, mp4_name = filename_part.split("__", 1) else: cam_name = "apcam" mp4_name = filename_part logger.info(f"[DEMAND SIGNAL] Camera='{cam_name}', Video='{mp4_name}'") # Resolve search candidates on local disk candidate_paths = [ WATCH_DIR / "motion" / cam_name / mp4_name, WATCH_DIR / cam_name / "motion" / mp4_name, WATCH_DIR / "motion" / mp4_name ] local_mp4 = None for path in candidate_paths: if path.is_file(): local_mp4 = path break # Global fallback search across WATCH_DIR if exact directory match fails if not local_mp4: matches = list(WATCH_DIR.glob(f"**/{mp4_name}")) if matches: local_mp4 = matches[0] target_key = f"{LOCATION_ID}/motion/{cam_name}/{mp4_name}" if local_mp4 and local_mp4.is_file(): logger.info(f"Uploading requested motion clip: {local_mp4} -> {target_key}") success = upload_file(local_mp4, target_key, "video/mp4") if success: try: s3_client.delete_object(Bucket=R2_BUCKET_NAME, Key=marker_key) logger.info(f"Cleared request ticket: {marker_key}") except Exception as ex: logger.warning(f"Failed to clear ticket {marker_key}: {ex}") else: logger.error(f"Requested clip '{mp4_name}' not found on local disk under {WATCH_DIR}") except Exception as ex: logger.error(f"Connection or polling error in motion listener: {ex}") time.sleep(1.5)
