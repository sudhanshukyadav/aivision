// ============================================================ // AISYS Live Camera Dashboard (4-Camera Grid & FIFO 4-Quad Stills) // ============================================================ const SNAPSHOT_POLL_INTERVAL_MS = 2500; let pollTimer = null; let hlsPlayers = {}; // Safely obtain session on DOM ready without immediate bounce-back loop document.addEventListener('DOMContentLoaded', () => { const sessionRaw = localStorage.getItem('aisys_session'); if (!sessionRaw) { window.location.replace('/login.html'); return; } let session; try { session = JSON.parse(sessionRaw); } catch (err) { localStorage.removeItem('aisys_session'); window.location.replace('/login.html'); return; } if (!session || !session.locationId || !session.token) { localStorage.removeItem('aisys_session'); window.location.replace('/login.html'); return; } // Update header text const userDisplay = document.getElementById('user-display'); if (userDisplay) { userDisplay.textContent = `${session.username} @ ${session.locationId} [${(session.role || 'viewer').toUpperCase()}]`; } // Fallback to 4 default camera slots if database only returned 1 camera let cameras = session.cameras || []; if (cameras.length === 0) { cameras = [{ camera_id: 'apcam', name: 'Camera 1' }]; } // Ensure 4 slots to fill the 2x2 camera grid cleanly while (cameras.length < 4) { cameras.push({ camera_id: `cam${cameras.length + 1}`, name: `Camera ${cameras.length + 1}`, isPlaceholder: true }); } render2x2Grid(session.locationId, cameras.slice(0, 4)); startFifoPolling(session.locationId, cameras.slice(0, 4)); }); // ------------------------------------------------------------ // RENDER 2x2 CAMERA GRID // ------------------------------------------------------------ function render2x2Grid(locationId, cameras) { const gridContainer = document.getElementById('camera-grid'); if (!gridContainer) return; gridContainer.innerHTML = cameras.map(cam => { const camId = cam.camera_id.toLowerCase(); const camName = cam.name || cam.camera_id; return `
${camName} ${locationId}
${[0, 1, 2, 3].map(idx => `
${idx === 0 ? 'Latest (T-0)' : `T-${idx}`}
Play 10s Clip
`).join('')}
`; }).join(''); } // ------------------------------------------------------------ // SLIDING WINDOW FIFO SNAPSHOT POLLER // ------------------------------------------------------------ function startFifoPolling(locationId, cameras) { if (pollTimer) clearInterval(pollTimer); pollTimer = setInterval(() => { const ts = Date.now(); cameras.forEach(cam => { const camId = cam.camera_id.toLowerCase(); const wrap = document.getElementById(`vwrap-${camId}`); // Do not refresh snapshots while user is actively watching a video or HLS stream if (wrap && wrap.style.display === 'block') return; // Update the 4 indices (0 = newest, 3 = oldest) for (let idx = 0; idx < 4; idx++) { const el = document.getElementById(`img-${camId}-${idx}`); if (el) { const freshSrc = `/api/snapshot/latest?location=${locationId}&cam=${camId}&index=${idx}&t=${ts}`; const buffer = new Image(); buffer.src = freshSrc; buffer.onload = () => { el.src = freshSrc; }; } } }); }, SNAPSHOT_POLL_INTERVAL_MS); } // ------------------------------------------------------------ // ON-DEMAND MOTION RETRIEVAL // ------------------------------------------------------------ async function onSnapshotClick(locationId, camId, index) { const quad = document.getElementById(`quad-${camId}`); const wrap = document.getElementById(`vwrap-${camId}`); const player = document.getElementById(`vplayer-${camId}`); const msg = document.getElementById(`vmsg-${camId}`); quad.style.display = 'none'; wrap.style.display = 'block'; msg.style.display = 'flex'; msg.textContent = 'Querying motion recording…'; try { const metaRes = await fetch(`/api/snapshot/latest?location=${locationId}&cam=${camId}&index=${index}&json=1`); if (!metaRes.ok) throw new Error('Snapshot record not found'); const meta = await metaRes.json(); const rawKey = meta.key || ''; const baseName = rawKey.split('/').pop().replace(/\.(jpe?g|png|webp)$/i, ''); const mp4Name = `${baseName}_motion.mp4`; msg.textContent = 'Requesting 10s video from camera drive…'; await fetch(`/api/motion/request?location=${locationId}&cam=${camId}&file=${encodeURIComponent(mp4Name)}`); let attempts = 0; const checkTimer = setInterval(async () => { attempts++; msg.textContent = `Transferring clip… (${attempts}s)`; try { const statusRes = await fetch(`/api/motion/status?location=${locationId}&cam=${camId}&file=${encodeURIComponent(mp4Name)}`); const statusData = await statusRes.json(); if (statusData.status === 'ready' || attempts >= 10) { clearInterval(checkTimer); msg.style.display = 'none'; player.src = `/api/motion/latest?location=${locationId}&cam=${camId}&file=${encodeURIComponent(mp4Name)}`; player.play().catch(() => {}); } } catch (e) { clearInterval(checkTimer); closeVideoPlayer(camId); } }, 1000); } catch (err) { msg.textContent = 'Motion clip unavailable for this frame.'; setTimeout(() => closeVideoPlayer(camId), 2000); } } // ------------------------------------------------------------ // HLS LIVE STREAMING // ------------------------------------------------------------ function toggleLiveHls(locationId, camId) { const quad = document.getElementById(`quad-${camId}`); const wrap = document.getElementById(`vwrap-${camId}`); const player = document.getElementById(`vplayer-${camId}`); const btn = document.getElementById(`btn-live-${camId}`); const msg = document.getElementById(`vmsg-${camId}`); if (wrap.style.display === 'block') { closeVideoPlayer(camId); return; } quad.style.display = 'none'; wrap.style.display = 'block'; msg.style.display = 'none'; btn.textContent = 'Return to Stills'; const hlsUrl = `/live/${camId}.m3u8?location=${locationId}`; if (player.canPlayType('application/vnd.apple.mpegurl')) { player.src = hlsUrl; player.play().catch(() => { msg.style.display = 'flex'; msg.textContent = 'Live stream offline.'; }); } else if (window.Hls && Hls.isSupported()) { if (hlsPlayers[camId]) hlsPlayers[camId].destroy(); const hls = new Hls(); hls.loadSource(hlsUrl); hls.attachMedia(player); hls.on(Hls.Events.MANIFEST_PARSED, () => player.play().catch(() => {})); hls.on(Hls.Events.ERROR, () => { msg.style.display = 'flex'; msg.textContent = 'Live stream offline.'; }); hlsPlayers[camId] = hls; } } function closeVideoPlayer(camId) { const quad = document.getElementById(`quad-${camId}`); const wrap = document.getElementById(`vwrap-${camId}`); const player = document.getElementById(`vplayer-${camId}`); const btn = document.getElementById(`btn-live-${camId}`); const msg = document.getElementById(`vmsg-${camId}`); if (player) { player.pause(); player.src = ''; } if (hlsPlayers[camId]) { hlsPlayers[camId].destroy(); delete hlsPlayers[camId]; } if (btn) btn.textContent = 'Live Stream'; if (msg) msg.style.display = 'none'; if (wrap) wrap.style.display = 'none'; if (quad) quad.style.display = 'grid'; } function logout() { localStorage.removeItem('aisys_session'); window.location.replace('/login.html'); }
